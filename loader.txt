-- Split Lobby/Game loader (single-instance guard, raw URLs, teleport-persistent, forced GAME after teleport)
-- Put in autoexec OR run once in the lobby.

local LOBBY_URL = "https://raw.githubusercontent.com/wtfx0rzz-mark/universal-tjhredhytjerwgfgh/main/main.lua"
local GAME_URL  = "https://raw.githubusercontent.com/wtfx0rzz-mark/aZ8rY2dLq4NfX1pT9sGv/main/main.lua"

-- Optional: if your scripts rely on _G.UI tabs existing, set these.
local LOBBY_TAB_NAME = ""
local GAME_TAB_NAME  = ""

local function env()
    return (getgenv and getgenv()) or _G
end

local function log(s)
    print("[SPLITBOOT] " .. tostring(s))
end

-- Single-instance guard:
-- Prevents multiple bootstraps from running concurrently in the same server/job.
local function acquireSingleton()
    local E = env()
    E.__SPLITBOOT_SINGLETON = E.__SPLITBOOT_SINGLETON or {}
    local key = tostring(game.JobId or "nojob")
    if E.__SPLITBOOT_SINGLETON[key] then
        log("Singleton already active for JobId=" .. key .. " (skipping)")
        return false
    end
    E.__SPLITBOOT_SINGLETON[key] = true
    return true
end

local function qot()
    return queue_on_teleport
        or (syn and syn.queue_on_teleport)
        or (fluxus and fluxus.queue_on_teleport)
        or (krnl and krnl.queue_on_teleport)
end

local function httpGet(url)
    if game and game.HttpGet then
        return game:HttpGet(url)
    end
    if syn and syn.request then
        local r = syn.request({ Url = url, Method = "GET" })
        if r and r.Body then return r.Body end
    end
    if request then
        local r = request({ Url = url, Method = "GET" })
        if r and r.Body then return r.Body end
    end
    if http_request then
        local r = http_request({ Url = url, Method = "GET" })
        if r and r.Body then return r.Body end
    end
    error("No HTTP fetch available (need game:HttpGet or executor request()).")
end

local function waitForGlobalTab(tabName, timeout)
    if not tabName or tabName == "" then return true end
    timeout = timeout or 25
    local t0 = os.clock()
    repeat
        local UI = rawget(_G, "UI")
        if UI and UI.Tabs and UI.Tabs[tabName] then
            return true
        end
        task.wait(0.15)
    until (os.clock() - t0) > timeout
    return false
end

local function waitCommonReady(timeout)
    timeout = timeout or 25
    if not game:IsLoaded() then game.Loaded:Wait() end
    local Players = game:GetService("Players")
    local RS = game:GetService("ReplicatedStorage")
    local lp = Players.LocalPlayer
    local t0 = os.clock()
    while not lp and (os.clock() - t0) < timeout do
        task.wait(0.1)
        lp = Players.LocalPlayer
    end
    if lp then
        pcall(function()
            lp:WaitForChild("PlayerGui", timeout)
        end)
    end
    pcall(function()
        RS:WaitForChild("RemoteEvents", timeout)
    end)
    return lp
end

local function runUrl(tag, url, tabName)
    waitCommonReady(25)
    if tabName and tabName ~= "" then
        local ok = waitForGlobalTab(tabName, 25)
        log(tag .. " waitForTab(" .. tabName .. ")=" .. tostring(ok))
    end
    log(tag .. " fetching: " .. url)
    local src = httpGet(url)
    local fn = loadstring(src)
    return fn()
end

local function ranKey(mode)
    return "__SPLITBOOT_RAN_" .. tostring(mode) .. "_" .. tostring(game.JobId)
end

local function runOnce(mode, fn)
    local E = env()
    local k = ranKey(mode)
    if E[k] then
        log(mode .. " already ran for this JobId")
        return
    end
    E[k] = true
    fn()
end

local function queueGameAfterTeleport()
    local q = qot()
    if not q then
        log("queue_on_teleport NOT available in this executor.")
        return
    end

    local gameTab = GAME_TAB_NAME or ""
    local url = GAME_URL

    local payload =
        "local function env() return (getgenv and getgenv()) or _G end\n" ..
        "local E=env()\n" ..
        "E.__SPLITBOOT_FORCE='GAME'\n" ..
        "E.__SPLITBOOT_SINGLETON = E.__SPLITBOOT_SINGLETON or {}\n" ..
        "E.__SPLITBOOT_SINGLETON[tostring(game.JobId or 'nojob')] = true\n" ..
        "local function log(s) print('[SPLITBOOT] '..tostring(s)) end\n" ..
        "local function httpGet(url)\n" ..
        "  if game and game.HttpGet then return game:HttpGet(url) end\n" ..
        "  if syn and syn.request then local r=syn.request({Url=url,Method='GET'}) if r and r.Body then return r.Body end end\n" ..
        "  if request then local r=request({Url=url,Method='GET'}) if r and r.Body then return r.Body end end\n" ..
        "  if http_request then local r=http_request({Url=url,Method='GET'}) if r and r.Body then return r.Body end end\n" ..
        "  error('No HTTP fetch available in post-teleport payload')\n" ..
        "end\n" ..
        "local function waitForGlobalTab(tabName, timeout)\n" ..
        "  if not tabName or tabName=='' then return true end\n" ..
        "  timeout = timeout or 25\n" ..
        "  local t0=os.clock()\n" ..
        "  repeat\n" ..
        "    local UI=rawget(_G,'UI')\n" ..
        "    if UI and UI.Tabs and UI.Tabs[tabName] then return true end\n" ..
        "    task.wait(0.15)\n" ..
        "  until (os.clock()-t0)>timeout\n" ..
        "  return false\n" ..
        "end\n" ..
        "if not game:IsLoaded() then game.Loaded:Wait() end\n" ..
        "local Players=game:GetService('Players')\n" ..
        "local RS=game:GetService('ReplicatedStorage')\n" ..
        "pcall(function() RS:WaitForChild('RemoteEvents', 25) end)\n" ..
        "local lp=Players.LocalPlayer\n" ..
        "local t0=os.clock()\n" ..
        "while not lp and (os.clock()-t0)<25 do task.wait(0.1) lp=Players.LocalPlayer end\n" ..
        "if lp then pcall(function() lp:WaitForChild('PlayerGui', 25) end) end\n" ..
        "log('POST-TELEPORT placeId='..tostring(game.PlaceId)..' jobId='..tostring(game.JobId)..' privateId='..tostring(game.PrivateServerId))\n" ..
        ("local tabName=%q\n"):format(gameTab) ..
        "if tabName~='' then log('GAME waitForTab('..tabName..')='..tostring(waitForGlobalTab(tabName,25))) end\n" ..
        ("local url=%q\n"):format(url) ..
        "log('GAME fetching: '..url)\n" ..
        "local src=httpGet(url)\n" ..
        "local fn=loadstring(src)\n" ..
        "fn()\n"

    local ok, err = pcall(function()
        q(payload)
    end)
    if ok then
        log("Queued GAME script for after teleport (forced GAME mode)")
    else
        log("queue_on_teleport threw: " .. tostring(err))
    end
end

if not game:IsLoaded() then game.Loaded:Wait() end
log("START placeId=" .. tostring(game.PlaceId) .. " jobId=" .. tostring(game.JobId) .. " privateId=" .. tostring(game.PrivateServerId))

if not acquireSingleton() then
    return
end

local E = env()
local forced = E.__SPLITBOOT_FORCE -- nil / "LOBBY" / "GAME"

if forced == "GAME" then
    runOnce("GAME", function()
        runUrl("GAME", GAME_URL, GAME_TAB_NAME)
    end)
    return
elseif forced == "LOBBY" then
    runOnce("LOBBY", function()
        runUrl("LOBBY", LOBBY_URL, LOBBY_TAB_NAME)
    end)
    return
end

-- Default behavior when started in lobby:
-- 1) Queue GAME for teleport destination
-- 2) Run LOBBY now
runOnce("QUEUE", function()
    queueGameAfterTeleport()
end)

runOnce("LOBBY", function()
    runUrl("LOBBY", LOBBY_URL, LOBBY_TAB_NAME)
end)
